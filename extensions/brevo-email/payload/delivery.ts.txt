import { z } from 'npm:zod@4.4.3';

export const recipientSchema = z.object({ email: z.email().max(254), name: z.string().max(200).regex(/^[^\r\n]*$/).optional() }).strict();
export const notificationSchema = z.object({
  id: z.uuid(),
  template: z.string().regex(/^[A-Z][A-Z0-9_]{0,63}$/),
  to: recipientSchema,
  params: z.record(z.string(), z.json()),
  metadata: z.record(z.string(), z.json()).optional(),
}).strict().refine(value => new TextEncoder().encode(JSON.stringify(value)).length <= 100_000);
export type Notification = z.infer<typeof notificationSchema>;
export type Database = { rpc: (name: string, args?: Record<string, unknown>) => PromiseLike<{ data?: unknown; error: unknown }> };

/** Trusted server/Edge code only. Use a service client scoped to api_edge.
 * Persist then dispatch immediately after commit; returned IDs are NOT proof of delivery.
 * Use the SQL enqueue function inside a business transaction when atomicity is required.
 */
export async function sendBrevoEmails(database: Database, notifications: Notification[]) {
  const parsed = z.array(notificationSchema).min(1).max(1000).parse(notifications);
  if (new Set(parsed.map(value => value.id)).size !== parsed.length) throw new Error('Duplicate notification IDs');
  const result = await database.rpc('ext_brevo_transactional_enqueue', { notifications: parsed });
  if (result.error || !z.array(z.uuid()).safeParse(result.data).success) throw new Error('Could not record notifications');
  return { recorded: result.data as string[] };
}
